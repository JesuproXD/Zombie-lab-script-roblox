-- Función para recopilar las partes del personaje que deben ser visibles
local function actualizarPartesPersonaje(character)
    ThirdPersonParts = {}
    if not character then return end

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") then
            if object.Name == "Head" or object.Parent:IsA("Accessory") then
                table.insert(ThirdPersonParts, object)
            end
        elseif object:IsA("Decal") and object.Name == "face" then
            table.insert(ThirdPersonParts, object)
        end
    end
end

-- Función principal para controlar la cámara
local function actualizarCamara()
    if thirdPersonActivo then
        Camera.CameraType = Enum.CameraType.Scriptable

        if not ThirdPersonConnection then
            ThirdPersonConnection = RunService.RenderStepped:Connect(function()
                local character = LocalPlayer.Character
                if not character then return end

                local hrp = character:FindFirstChild("HumanoidRootPart")
                if not hrp then return end

                local offsetVector = Vector3.new(offsetX, offsetY, offsetZ)

                local camPos = hrp.CFrame:ToWorldSpace(CFrame.new(offsetVector))
                local targetCFrame = CFrame.new(camPos.Position, hrp.Position)

                Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, 0.2)

                -- Forzar visibilidad de la cabeza y accesorios
                for _, part in ipairs(ThirdPersonParts) do
                    if part and part.Parent then
                        if part:IsA("BasePart") then
                            part.LocalTransparencyModifier = 0
                        elseif part:IsA("Decal") then
                            part.Transparency = 0
                        end
                    end
                end
            end)
        end
    else
        -- Desactivar y restaurar la cámara normal
        if ThirdPersonConnection then
            ThirdPersonConnection:Disconnect()
            ThirdPersonConnection = nil
        end

        Camera.CameraType = Enum.CameraType.Custom

        for _, part in ipairs(ThirdPersonParts) do
            if part and part.Parent then
                if part:IsA("BasePart") then
                    part.LocalTransparencyModifier = 0
                end
            end
        end
    end
end

local itemsExcluidos = {

    ["PurchasableItem2"] = true,

    ["PurchasableItem4"] = true

-- Función auxiliar para verificar si un texto contiene algún término ignorado
local function esTextoIgnorado(texto)
    if not texto or texto == "" then return false end
    local textoLower = string.lower(texto)
    
    for nombreIgnorado, activo in pairs(ignoradosSeleccionados) do
        if activo and string.find(textoLower, string.lower(nombreIgnorado), 1, true) then
            return true
        end
    end
    return false
end

-- Función principal de activación
local function triggerPrompt(prompt)
    if not prompt or not prompt.Enabled or not prompt.Parent then return end

    local promptName = prompt.Name
    local parentName = prompt.Parent.Name

    -- 1. Regla específica: Excluir ModulePrompt si su abuelo es PurchasableItem1..4
    if promptName == "ModulePrompt" and prompt.Parent.Parent then
        local grandParentName = prompt.Parent.Parent.Name
        if itemsExcluidos[grandParentName] then
            return -- Se ignora la interacción
        end
    end

    -- 2. Verificación directa por Nombre del Prompt o de su Padre (Modelo/Parte)
    if ignoradosSeleccionados[promptName] or ignoradosSeleccionados[parentName] then
        return
    end

    -- 3. Verificación inteligente por Texto Visual (ObjectText / ActionText)
    if esTextoIgnorado(prompt.ObjectText) or esTextoIgnorado(prompt.ActionText) then
        return
    end

    -- Ejecución de la interacción
    if typeof(fireproximityprompt) == "function" then
        fireproximityprompt(prompt)
    else
        prompt:InputHoldBegin()
        if prompt.HoldDuration > 0 then
            task.wait(prompt.HoldDuration)
        end
        prompt:InputHoldEnd()
    end
end

-- Escuchar eventos de apariciones de prompts
ProximityPromptService.PromptShown:Connect(function(prompt)
    if autoInteractActivo and prompt.Enabled then
        task.spawn(triggerPrompt, prompt)
    end
end)

local instantInteractActivo = false
local conexionDescendantAdded = nil

-- Tabla para memorizar la duración original de cada prompt sin borrarlos
local duracionesOriginales = {}

-- Cambia solo el tiempo de espera a 0 (mantiene el prompt activado y existiendo)
local function aplicarInstantaneo(prompt)
	if not prompt:IsA("ProximityPrompt") then return end
	
	if not duracionesOriginales[prompt] then
		duracionesOriginales[prompt] = prompt.HoldDuration
	end
	
	prompt.HoldDuration = 0
end

-- Devuelve el tiempo original al prompt sin alterar nada más
local function restaurarInstantaneo(prompt)
	if duracionesOriginales[prompt] and prompt.Parent then
		prompt.HoldDuration = duracionesOriginales[prompt]
	end
end

-- Tu Toggle de la UI
TabBox2:AddToggle("InstantInteact", {
	Text = "Interacción instantánea",
	Tooltip = "El tiempo de activación de los prompts es instantánea",

	Default = false,
	Disabled = false,
	Visible = true,
	Risky = false,

	Callback = function(Value)
		instantInteractActivo = Value
		
		if instantInteractActivo then
			-- 1. Cambiamos el tiempo a los que ya están cargados en el mapa
			for _, objeto in pairs(workspace:GetDescendants()) do
				if objeto:IsA("ProximityPrompt") then
					aplicarInstantaneo(objeto)
				end
			end
			
			-- 2. Cambiamos el tiempo a los nuevos que vayan apareciendo en nuevas habitaciones
			conexionDescendantAdded = workspace.DescendantAdded:Connect(function(descendant)
				if instantInteractActivo and descendant:IsA("ProximityPrompt") then
					-- Un pequeño delay asegura que el script del juego no sobreescriba nuestro cambio
					task.defer(aplicarInstantaneo, descendant)
				end
			end)
		else
			-- Desconectamos el detector de nuevas salas
			if conexionDescendantAdded then
				conexionDescendantAdded:Disconnect()
				conexionDescendantAdded = nil
			end
			
			-- Restauramos los segundos originales uno a uno
			for prompt, _ in pairs(duracionesOriginales) do
				restaurarInstantaneo(prompt)
			end
			
			-- Vaciamos la lista de memoria
			table.clear(duracionesOriginales)
		end
	end,
})

-- Modificamos la función para que reciba el estado que queremos ponerle al CanTouch
local function RevisarPinchos(estadoCanTouch)
    -- Si la función se ejecuta por el evento .Changed, estadoCanTouch será el número de habitación.
    -- Por eso, si no es un booleano (true/false), forzamos a que sea false (AntiPinchos Activo).
    if type(estadoCanTouch) ~= "boolean" then
        estadoCanTouch = false
    end

    if workspace:FindFirstChild("CurrentRooms") then
        for _, objeto in ipairs(workspace:FindFirstChild("CurrentRooms"):GetDescendants()) do
            if objeto.Name == "Snare" then
                local hitbox = objeto:FindFirstChild("Hitbox")
                if hitbox and hitbox:IsA("BasePart") then
                    hitbox.CanTouch = estadoCanTouch
                end
            end
        end
    end
end

            end
            
            -- 2. Volvemos a ACTIVAR el CanTouch de todos los Snares actuales (pasamos true)
            RevisarPinchos(true)
        end
    end,
})

-- Función auxiliar para verificar si la habitación actual es la 50
local function estaEnHabitacion50()
    local currentRooms = workspace:FindFirstChild("CurrentRooms")
    if currentRooms and currentRooms:FindFirstChild("50") then
        return true
    end
    return false
end

-- Función auxiliar para obtener el nombre de la figura según su ImageRectOffset.X
local function obtenerNombreFigura(imageLabel)
    if not imageLabel or not imageLabel:IsA("ImageLabel") then return nil end
    local offsetX = math.round(imageLabel.ImageRectOffset.X)
    return DiccionarioFiguras[offsetX]
end

-- Función principal que calcula la combinación y genera las alertas
local function procesarCodigoBiblioteca(forzarNotificacion)
    if not AutocompletarActivo or not estaEnHabitacion50() then return end

    local localPlayer = game:GetService("Players").LocalPlayer
    
    -- Paso A: Obtener los números recolectados de los libros (Hints)
    local hintsFolder = localPlayer.PlayerGui:FindFirstChild("PermUI") and localPlayer.PlayerGui.PermUI:FindFirstChild("Hints")
    if not hintsFolder then return end

    local librosTraducidos = {}
    for _, item in ipairs(hintsFolder:GetChildren()) do
        if item:IsA("ImageLabel") and item.Visible then
            local figura = obtenerNombreFigura(item)
            local textLabel = item:FindFirstChild("TextLabel")
            
            if figura and textLabel then
                local numero = tonumber(textLabel.Text)
                if numero then
                    librosTraducidos[figura] = numero
                end
            end
        end
    end

    -- Paso B: Buscar la hoja de pistas específicamente en el Personaje (Equipada)
    local personaje = localPlayer.Character
    local hoja = personaje and personaje:FindFirstChild("LibraryHintPaper")
    
    local uiHoja = hoja and hoja:FindFirstChild("UI")
    if not uiHoja then return end

    -- CORRECCIÓN AQUÍ: 'codigoFinal' ahora es LOCAL y empieza limpio ("") cada vez que se escanea la hoja
    local codigoFinal = ""
    local faltanLibros = false

    -- Paso C: Descifrar el orden de las figuras desde la hoja (del 1 al 5)
    for i = 1, 5 do
        local ranuraHoja = uiHoja:FindFirstChild(tostring(i))
        if ranuraHoja and ranuraHoja:IsA("ImageLabel") then
            local figuraRequerida = obtenerNombreFigura(ranuraHoja)
            
            if figuraRequerida then
                local numeroAsignado = librosTraducidos[figuraRequerida]
                if numeroAsignado then
                    codigoFinal = codigoFinal .. tostring(numeroAsignado)
                else
                    codigoFinal = codigoFinal .. "_"
                    faltanLibros = true
                end
            else
                codigoFinal = codigoFinal .. "?"
            end
        end
    end

    -- Paso D: Enviar notificación si el código cambió o si se forzó al equipar la hoja
    if (codigoFinal ~= UltimoCodigoNotificado and codigoFinal ~= "") or forzarNotificacion then
        UltimoCodigoNotificado = codigoFinal
        
        if faltanLibros then
            Library:Notify("Código: " .. codigoFinal)
            TenemosCodigo = false
        else
            Library:Notify("Código: " .. codigoFinal)
            -- Guardamos el código limpio en una variable global compartida sólo si está completo
            _G.CodigoPuerta50Guardado = codigoFinal 
            TenemosCodigo = true
        end
        
        if SonidoNotificacion then
            SonidoNotificacion:Play()
        end
    end
end

-- Función para manejar el apagado del Highlight molesto de la puerta
local function limpiarHighlightPuerta()
    local currentRooms = workspace:FindFirstChild("CurrentRooms")
    local room50 = currentRooms and currentRooms:FindFirstChild("50")
    local puerta = room50 and room50:FindFirstChild("Door")
    
    if puerta then
        for _, objeto in ipairs(puerta:GetDescendants()) do
            if objeto:IsA("Highlight") and objeto.Enabled then
                objeto.Enabled = false
            end
        end
    end
end

-- Función para conectar/desconectar el evento del candado
local function gestionarConexionPrompt(activar)
    if activar then
        if ConexionPrompt then ConexionPrompt:Disconnect() end
        
        task.spawn(function()
            while AutocompletarActivo do
                local currentRooms = workspace:FindFirstChild("CurrentRooms")
                local room50 = currentRooms and currentRooms:FindFirstChild("50")
                local padlock = room50 and room50:FindFirstChild("Door") and room50.Door:FindFirstChild("Padlock")
                local prompt = padlock and padlock:FindFirstChild("ActivateEventPrompt")
                
                if prompt and prompt:IsA("ProximityPrompt") then
                    ConexionPrompt = prompt.Triggered:Connect(function(player)
                        if player == game:GetService("Players").LocalPlayer then
                            limpiarHighlightPuerta()
                        end
                    end)
                    break 
                end
                task.wait(1)
            end
        end)
    else
        if ConexionPrompt then
            ConexionPrompt:Disconnect()
            ConexionPrompt = nil
        end
    end
end

-- Variables de configuración física
local ALTURA_OBJETIVO = 100 
local VELOCIDAD_SUBIDA = 25 

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local JugadorLocal = Players.LocalPlayer

local VelocidadHumanoidOriginal = JugadorHumanoid.WalkSpeed

-- Función interna para subir al personaje (CORREGIDA)
local function SubirPersonaje()
	local Character = workspace:FindFirstChild(JugadorLocal.Name)
	if not Character then return end
	
	local HRP = Character:FindFirstChild("HumanoidRootPart")
	local Humanoid = Character:FindFirstChild("Humanoid")
	if not HRP or not Humanoid or HRP.Anchored then return end
	
	local PosOriginalCFrame = HRP.CFrame
	local PosOriginalVector = HRP.Position
	Humanoid.PlatformStand = true

	-- [CORRECCIÓN] Declaramos las variables locales correctamente aquí dentro para el bucle
	local posicionActualY = PosOriginalVector.Y
	local alturaFinalY = PosOriginalVector.Y + ALTURA_OBJETIVO

	local conexionMovimiento
	conexionMovimiento = RunService.Heartbeat:Connect(function(dt)
		-- Verificación por si el personaje muere o se elimina en plena subida
		if not HRP or not HRP.Parent then 
			conexionMovimiento:Disconnect() 
			return 
		end
		
        Humanoid.WalkSpeed = 0

		local distanciaAMover = VELOCIDAD_SUBIDA * dt
		posicionActualY = math.min(posicionActualY + distanciaAMover, alturaFinalY)
		
		HRP.CFrame = CFrame.new(HRP.Position.X, posicionActualY, HRP.Position.Z) * PosOriginalCFrame.Rotation
		
		if posicionActualY >= alturaFinalY then
			conexionMovimiento:Disconnect()
		end
	end)

	-- Esperamos a que termine de subir
	repeat task.wait() until posicionActualY >= alturaFinalY or not modoDiosActivo
	
	-- Si desactivaste el toggle mientras subía, cancelamos el anclaje
	if not modoDiosActivo then return end

	task.wait(0.5) -- Sincronización de red

	HRP.Anchored = true
	Humanoid.PlatformStand = false
end

	local Character = JugadorLocal.Character or workspace:FindFirstChild(JugadorLocal.Name)
	local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
	if HRP then
		HRP.Anchored = false
        JugadorHumanoid.WalkSpeed = VelocidadHumanoidOriginal
	end
end

-- Variables de control
local antiObstruccionesActivo = false
local conexionSeek = nil
local partesModificadas = {}

-- Función para restaurar las partes a su estado original
local function restaurarPartes()
    for part, canTouchOriginal in pairs(partesModificadas) do
        if part and part.Parent then
            part.CanTouch = canTouchOriginal
        end
    end
    partesModificadas = {}
end

-- Función principal para procesar las partes dañinas/obstrucciones
local function procesarObstruccionesSeek()
    if not antiObstruccionesActivo then
        restaurarPartes()
        return
    end

    local currentRooms = Workspace:FindFirstChild("CurrentRooms")
    if not currentRooms then return end

    -- Verificamos si existe el evento de la persecución de Seek en la habitación
    local seekTriggerFound = false
    for _, desc in ipairs(currentRooms:GetDescendants()) do
        if desc.Name == "TriggerSeek" then
            seekTriggerFound = true
            break
        end
    end

    -- Si se detecta la persecución de Seek, modificamos las HurtPart
    if seekTriggerFound then
        for _, desc in ipairs(currentRooms:GetDescendants()) do
            if desc.Name == "HurtPart" and desc:IsA("BasePart") then
                if partesModificadas[desc] == nil then
                    partesModificadas[desc] = desc.CanTouch
                end
                desc.CanTouch = false
            end
        end
    end
end

            conexionSeek:Disconnect()
            conexionSeek = nil
        end

        if Value then
            -- Ejecución inmediata por si ya estamos en la sala de Seek

local function aplicarJugadorESP(player)
    -- 🛑 EXCLUSIÓN: Si es el jugador local, no hacer absolutamente nada
    if player == localPlayer then return end

    local function enCharacterAdded(character)
        task.wait(0.1) -- Espera corta a que carguen las extremidades
        if not character:IsDescendantOf(workspace) then return end

        -- Guardamos el personaje en la caché
        jugadoresActivos[player] = character

        -- 🔴 SI SE DESACTIVA EL TOGGLE: Eliminar Highlight de inmediato
        if not _G.JugadoresESP then
            local hViejo = character:FindFirstChild("JugadorHighlight")
            if hViejo then hViejo:Destroy() end
            return
        end

        -- 🛠️ INYECCIÓN / ACTUALIZACIÓN DEL HIGHLIGHT
        local h = character:FindFirstChild("JugadorHighlight")
        if not h then
            h = Instance.new("Highlight")
            h.Name = "JugadorHighlight"
            h.OutlineColor = Color3.fromRGB(0, 0, 0) -- Contorno negro para resaltar el blanco
            h.FillTransparency = 0.5
            h.OutlineTransparency = 0
            h.Adornee = character
            h.Parent = character
        end

        h.Enabled = _G.JugadoresESP
        h.FillColor = _G.ColorJugadores
    end

    -- Conectar si el personaje ya existe o cuando reaparezca
    if player.Character then
        task.spawn(enCharacterAdded, player.Character)
    end
    player.CharacterAdded:Connect(enCharacterAdded)
end

    for _, player in pairs(Players:GetPlayers()) do

end

local function buscarParteBase(puertaModel)
    if not puertaModel then return nil end
    local hit = puertaModel:FindFirstChild("Door") or puertaModel:FindFirstChild("Panel") or puertaModel:FindFirstChild("Frame")
    if hit and hit:IsA("BasePart") then return hit end
    return puertaModel:FindFirstChildWhichIsA("BasePart", true)
end

local function limpiarVisualesPuertas()
    puertaActualModelo = nil
    parteFisicaActual = nil
    
    -- Limpieza pasiva total en nuestro contenedor seguro
    for _, obj in ipairs(ScreenGui:GetChildren()) do
        if obj.Name == "SecureHighlight" or obj.Name == "DoorTexto" then
            obj:Destroy()
        end
    end
end

local function aplicarVisualesPuerta(doorModel)
    if not doorModel or not doorModel:IsA("Model") then return end
    if doorModel.Name:find("Dupe") or doorModel.Name == "DoorFake" then return end
    
    -- 👁️ MANIPULACIÓN SEGURA DE HIDDEN: Opacidad a 1 para evadir el Anticheat
    pcall(function()
        local hiddenObj = doorModel:FindFirstChild("Hidden")
        if hiddenObj then
            if hiddenObj:IsA("BasePart") then
                hiddenObj.Transparency = 1
            else
                for _, p in ipairs(hiddenObj:GetDescendants()) do
                    if p:IsA("BasePart") then p.Transparency = 1 end
                end
            end
        end
    end)

    if doorModel:FindFirstChild("Sign") then
        local parteObjetivo = buscarParteBase(doorModel)
        if not parteObjetivo then return end
        
        puertaActualModelo = doorModel
        parteFisicaActual = parteObjetivo

        -- 🟢 HIGHLIGHT (CONTORNO INDEPENDIENTE)
        local h = ScreenGui:FindFirstChild("SecureHighlight")
        if not PuertasESP_Activo then
            if h then h:Destroy() end
        else
            if not h then
                h = Instance.new("Highlight")
                h.Name = "SecureHighlight"
                h.OutlineColor = Color3.fromRGB(255, 255, 255)
                h.FillTransparency = 0.5
                h.OutlineTransparency = 0
                h.Parent = ScreenGui 
            end
            h.Adornee = doorModel 
            h.Enabled = true
            h.FillColor = ColorPuertas
        end
    end
end

local function actualizarHabitacionActual()
    -- Guardamos si el Highlight estaba prendido antes de resetear
    local prevHighlight = PuertasESP_Activo
    
    limpiarVisualesPuertas()
    
    PuertasESP_Activo = prevHighlight
    
    local rooms = workspace:FindFirstChild("CurrentRooms")
    if not rooms then return end

    local salaTarget = HabitacionActual and HabitacionActual.Value
    if not salaTarget then return end

    local roomActual = rooms:FindFirstChild(tostring(salaTarget))
    if not roomActual then return end

    pcall(function()
        local puertaEncontrada = nil
        for _, desc in ipairs(roomActual:GetDescendants()) do
            if desc.Name == "Door" and desc:IsA("Model") and desc:FindFirstChild("Sign") then
                puertaEncontrada = desc
                break
            end
        end

        if puertaEncontrada then
            aplicarVisualesPuerta(puertaEncontrada)
        end
    end)
end

-- 🔮 BUCLE DE RENDERIZADO (TEXTO + TAMAÑO DINÁMICO COMPATIBLE CON SLIDER)

local function iniciarBucleRenderizado()
    if scriptBucleActivo then return end
    scriptBucleActivo = true
    
    task.spawn(function()
        while scriptBucleActivo do
            pcall(function()
                local personaje = JugadorLocal.Character
                local raiz = personaje and personaje:FindFirstChild("HumanoidRootPart")

                local textoMaestroEncendido = Toggles.TextoESP and Toggles.TextoESP.Value
                local puertasSeleccionadas = _G.TextoESP_Config and _G.TextoESP_Config["Puertas"] == true

                if textoMaestroEncendido and puertasSeleccionadas and puertaActualModelo and puertaActualModelo.Parent and parteFisicaActual and parteFisicaActual.Parent then
                    
                    local bbGui = ScreenGui:FindFirstChild("DoorTexto")
                    if not bbGui then
                        bbGui = Instance.new("BillboardGui")
                        bbGui.Name = "DoorTexto"
                        bbGui.Size = UDim2.new(0, 200, 0, 40)
                        bbGui.AlwaysOnTop = true
                        bbGui.Parent = ScreenGui
                    end
                    
                    if bbGui.Adornee ~= parteFisicaActual then
                        bbGui.Adornee = parteFisicaActual
                    end

                    local label = bbGui:FindFirstChild("Label")
                    if not label then
                        label = Instance.new("TextLabel")
                        label.Name = "Label"
                        label.Size = UDim2.new(1, 0, 1, 0)
                        label.Font = Enum.Font.SourceSansBold
                        label.TextStrokeTransparency = 0
                        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        label.BackgroundTransparency = 1
                        label.Parent = bbGui
                    end

                    label.TextColor3 = ColorPuertas
                    
                    -- 🌟 VINCULACIÓN CON EL SLIDER DE TAMAÑO:
                    -- Intentamos leer Options.Tamano.Value (o Toggles.Tamano.Value según LinoriaLib). Si no existe, usa 20 por defecto.
                    local tamanoMenu = Options and Options.Tamano and Options.Tamano.Value or 20
                    label.TextSize = tamanoMenu

                    if raiz then
                        local numeroSalaBase = HabitacionActual and tonumber(HabitacionActual.Value) or 0
                        local numeroSiguientePuerta = numeroSalaBase + 1
                        local distancia = (raiz.Position - parteFisicaActual.Position).Magnitude
                        
                        label.Text = string.format("Puerta %d [%.1f m]", numeroSiguientePuerta, distancia)
                        label.Visible = true
                    end
                else
                    local bbGui = ScreenGui:FindFirstChild("DoorTexto")
                    if bbGui then bbGui:Destroy() end
                end
            end)
            task.wait(0.08)
        end
    end)
end

-- 🔮 LÓGICA DEL ESP CORREGIDA (CERO TEXTOS AQUÍ PARA EVITAR INVISIBILIDAD)
-- =============================================================================
local function aplicarHighlightLimpio(modelo)
    if not modelo:IsA("Model") then return end
    task.wait(0.1) -- Espera corta de carga
    
    -- 1. Buscamos el nombre limpio que usa tu menú (ej: "Figure")
    local nombreMenu = nombresJuego[modelo.Name]
    if not nombreMenu then return end

    -- Determinamos cuál es el objeto interno que llevará los efectos visuales
    local objetivoVisual = modelo
    
    if modelo.Name == "RushMoving" then
        local bloqueInterno = modelo:FindFirstChild("RushMoving") or modelo:FindFirstChildWhichIsA("BasePart", true)
        if bloqueInterno then
            if (_G.EntidadesESP and _G.EntidadesSeleccionadas[nombreMenu]) or (_G.TextoESP_Config and _G.TextoESP_Config["Entidades"]) then
                bloqueInterno.Transparency = 0
            end
            objetivoVisual = bloqueInterno
        end
    elseif modelo.Name == "AmbushMoving" then
        local bloqueInterno = modelo:FindFirstChild("AmbushMoving") or modelo:FindFirstChildWhichIsA("BasePart", true)
        if bloqueInterno then
            if (_G.EntidadesESP and _G.EntidadesSeleccionadas[nombreMenu]) or (_G.TextoESP_Config and _G.TextoESP_Config["Entidades"]) then
                bloqueInterno.Transparency = 0
            end
            objetivoVisual = bloqueInterno
        end
    elseif modelo.Name == "Eyes" then
        local core = modelo:FindFirstChild("Core")
        if core and core:IsA("BasePart") then
            if (_G.EntidadesESP and _G.EntidadesSeleccionadas[nombreMenu]) or (_G.TextoESP_Config and _G.TextoESP_Config["Entidades"]) then
                core.Transparency = 0
            end
            objetivoVisual = core
        end
    elseif modelo.Name == "FigureRig" then
        local root = modelo:FindFirstChild("Root")
        if root and root:IsA("BasePart") then root.Transparency = 1 end
    end

    -- Guardamos la referencia física en la caché para que el nuevo bucle maestro la controle
    entidadesActivas[modelo] = objetivoVisual

    --------------------------------------------------------
    -- 🛑 GESTIÓN INDEPENDIENTE DEL HIGHLIGHT (CONTORNO) --
    --------------------------------------------------------
    local h = objetivoVisual:FindFirstChild("EntidadHighlight")
    
    if not _G.EntidadesESP or _G.EntidadesSeleccionadas[nombreMenu] ~= true then 
        if h then h:Destroy() end
    else
        if not h then
            h = Instance.new("Highlight")
            h.Name = "EntidadHighlight"
            h.OutlineColor = Color3.fromRGB(255, 255, 255) 
            h.FillTransparency = 0.4
            h.OutlineTransparency = 0
            h.Adornee = objetivoVisual 
            h.Parent = objetivoVisual
        end
        h.Enabled = true
        h.FillColor = _G.ColorEntidades
    end
    --------------------------------------------------------
    -- 🧹 LIMPIEZA FINAL DE TRANSPARENCIAS               --
    --------------------------------------------------------
    local tieneHighlight = objetivoVisual:FindFirstChild("EntidadHighlight") ~= nil
    local tieneTexto = objetivoVisual:FindFirstChild("EntidadTexto") ~= nil

    if not tieneHighlight and not tieneTexto then
        if modelo.Name == "RushMoving" or modelo.Name == "AmbushMoving" or modelo.Name == "Eyes" then
            objetivoVisual.Transparency = 1
        end
    end
end

                        bbGui.Parent = objetivo
                    end
                    
                    if bbGui.AlwaysOnTop ~= true then bbGui.AlwaysOnTop = true end
                    if bbGui.Enabled ~= true then bbGui.Enabled = true end
                    
                    -- 2. Buscamos o creamos el ÚNICO TextLabel integrado
                    local label = bbGui:FindFirstChild("Label")
                    if not label then
                        label = Instance.new("TextLabel")
                        label.Name = "Label"
                        label.Size = UDim2.new(1, 0, 1, 0)
                        label.Font = Enum.Font.SourceSansBold
                        label.TextStrokeTransparency = 0       
                        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        label.Parent = bbGui
                    end
                    
                    -- Quitar cualquier tipo de ocultación externa
                    if label.Visible ~= true then label.Visible = true end
                    if label.BackgroundTransparency ~= 1 then label.BackgroundTransparency = 1 end
                    if label.TextTransparency ~= 0 then label.TextTransparency = 0 end
                    
                    -- 🌟 VINCULACIÓN CON EL SLIDER EN CALIENTE:
                    -- Intentamos leer el valor actual del slider "Tamano" en cada fotograma
                    local tamanoMenu = Options and Options.Tamano and Options.Tamano.Value or 20
                    if label.TextSize ~= tamanoMenu then
                        label.TextSize = tamanoMenu
                    end
                    
                    -- 3. Sincronizar texto unificado y color en tiempo real
                    if label then
                        local parteEntidad = modelo:FindFirstChildWhichIsA("BasePart", true)
                        if parteEntidad then
                            local nombreAMostrar = nombresPersonalizados[nombreMenu] or nombreMenu or "Entidad"
                            
                            if label.TextColor3 ~= _G.ColorEntidades then
                                label.TextColor3 = _G.ColorEntidades
                            end

                            local distancia = miRaiz and (miRaiz.Position - parteEntidad.Position).Magnitude or 0
                            local textoFinal = string.format("%s [%.1f]", nombreAMostrar, distancia)

                            if label.Text ~= textoFinal then
                                label.Text = textoFinal
                            end
                        end
                    end
                else
                    entidadesActivas[modelo] = nil 
                end
            end
        else
            -- Si se apaga, borramos instantáneamente todos los textos
            for _, objetivo in pairs(entidadesActivas) do
                if objetivo then
                    local bbGui = objetivo:FindFirstChild("EntidadTexto")
                    if bbGui then bbGui:Destroy() end
                end
            end
        end
    end
end)

-- Función para lanzar las alertas según los toggles activos
local function lanzarAlertaEntidad(entidad)
	-- 1. Filtros esenciales de seguridad básicos
	if not EntidadesSeleccionadas or not EntidadesSeleccionadas[entidad] then return end
	if not AlertaNotificacionActiva and not AlertaChatActiva then return end

	-- 2. Formateo del mensaje (con la regla del vacío/emojis)
	local sufijoMensaje = TextoPersonalizado
	if not sufijoMensaje or sufijoMensaje:gsub(" ", "") == "" then
		sufijoMensaje = "⚠️ " .. entidad .. " ha spawneado!"
	else
		sufijoMensaje = "⚠️ "..entidad .. " " .. sufijoMensaje
	end
	local mensajeDeAlerta = sufijoMensaje

	-- =============================================================================
	-- 🔊 + 🖥️ AUDIO Y NOTIFICACIÓN VISUAL (COMPROBACIÓN ULTRA SEGURA ANTICRASHEO)
	-- =============================================================================
	if AlertaNotificacionActiva then
		
		-- [PROTECCIÓN DE CARGA]: Verificamos rigurosamente que la UI de Linoria ya exista en memoria
		if SonidoNotificacion then
			-- Intentamos extraer de forma segura el valor del cuadro de texto sin romper el script si aún no carga
			local idEscrita = nil
			pcall(function()
				if Options and Options.SonidoDeNotificacion and Options.SonidoDeNotificacion.Value then
					idEscrita = Options.SonidoDeNotificacion.Value
				end
			end)

			-- Si logramos leer un número válido en la caja, se lo aplicamos al sonido principal
			if idEscrita and idEscrita ~= "" then
				SonidoNotificacion.SoundId = "rbxassetid://" .. tostring(idEscrita)
			end
			
			-- Reproduce el audio configurado
			SonidoNotificacion:Play()
		end

		-- Lanzamos inmediatamente la notificación visual en pantalla (Usa pcall por si la librería no ha terminado de iniciar)
		pcall(function()
			if Library and Library.Notify then
				Library:Notify(mensajeDeAlerta, 5)
			end
		end)
	end

	-- =============================================================================
	-- 💬 ALERTA DE CHAT PÚBLICO
	-- =============================================================================
	if AlertaChatActiva then
		local TextChatService = game:GetService("TextChatService")
		if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
			local generalChannel = TextChatService:FindFirstChild("TextChannels") and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
			if generalChannel then
				generalChannel:SendAsync(mensajeDeAlerta)
			end
		else
			local replicatedStorage = game:GetService("ReplicatedStorage")
			local defaultChatSystemChatEvents = replicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
			local sayMessageRequest = defaultChatSystemChatEvents and defaultChatSystemChatEvents:FindFirstChild("SayMessageRequest")
			if sayMessageRequest then
				sayMessageRequest:FireServer(mensajeDeAlerta, "All")
			end
		end
	end
end

local function procesarDeteccionSegura(objeto)
	-- 🛑 SI AMBOS TOGGLES ESTÁN APAGADOS, SE DETIENE LA DETECCIÓN DE INMEDIATO
	if not AlertaNotificacionActiva and not AlertaChatActiva then 
		return 
	end

	if not objeto:IsA("Model") then return end

	-- Comprobamos si el modelo que apareció coincide con un monstruo real de Doors
	local entidadMenu = nombresMonstruosJuego[objeto.Name]
	
	if entidadMenu then
		-- Comprobación: ¿La entidad está activa en tu Dropdown del menú?
		if EntidadesSeleccionadas and EntidadesSeleccionadas[entidadMenu] == true then
			
			-- Filtro anti-duplicados y anti-spam
			if objetosProcesados[objeto] or (objeto.Parent and objetosProcesados[objeto.Parent]) then 
				return 
			end
			
			-- Registramos la instancia física
			objetosProcesados[objeto] = true
			
			local conexionDestroy
			conexionDestroy = objeto.Destroying:Connect(function()
				objetosProcesados[objeto] = nil
				if conexionDestroy then conexionDestroy:Disconnect() end
			end)
			
			-- Enviamos a la función de alertas con el nombre correcto para el menú
			lanzarAlertaEntidad(entidadMenu)
		end
	end
end

-- Variables de control
local activado = false
local nivelTransparencia = 0.5
local ultimoEscondite = nil
local transparenciasOriginales = {}
local hidingConnection = nil

-- Referencia para el Slider
local SliderTransparenciaObj = nil

-- Función auxiliar para restaurar la transparencia original
local function restaurarEscondite()
    if ultimoEscondite then
        for part, transOriginal in pairs(transparenciasOriginales) do
            if part and part.Parent then
                part.Transparency = transOriginal
            end
        end
        transparenciasOriginales = {}
        ultimoEscondite = nil
    end
end

-- Función principal para aplicar la transparencia
local function actualizarTransparenciaEscondite()
    if not activado then
        restaurarEscondite()
        return
    end

    local jugadorFisico = LocalPlayer.Character
    if not jugadorFisico then 
        restaurarEscondite()
        return 
    end

    -- Verificamos el atributo "Hiding"
    local estaEscondido = jugadorFisico:GetAttribute("Hiding")

    if estaEscondido then
        local currentRooms = Workspace:FindFirstChild("CurrentRooms")
        if not currentRooms then return end

        local esconditeEncontrado = nil

        -- Buscamos el escondite actual del jugador
        for _, room in ipairs(currentRooms:GetChildren()) do
            for _, objeto in ipairs(room:GetDescendants()) do
                local hiddenPlayer = objeto:FindFirstChild("HiddenPlayer")
                if hiddenPlayer then
                    local val = hiddenPlayer.Value
                    if tostring(val) == LocalPlayer.Name or val == LocalPlayer.UserId or val == LocalPlayer or val == jugadorFisico then
                        esconditeEncontrado = objeto:IsA("Model") and objeto or objeto.Parent
                        break
                    end
                end
            end
            if esconditeEncontrado then break end
        end

        if esconditeEncontrado then
            if ultimoEscondite and ultimoEscondite ~= esconditeEncontrado then
                restaurarEscondite()
            end

            ultimoEscondite = esconditeEncontrado

            -- Guardamos transparencias originales y aplicamos la del slider
            for _, part in ipairs(esconditeEncontrado:GetDescendants()) do
                if part:IsA("BasePart") then
                    if transparenciasOriginales[part] == nil then
                        transparenciasOriginales[part] = part.Transparency
                    end
                    part.Transparency = nivelTransparencia
                end
            end
        end
    else
        -- Si dejó de esconderse, restauramos
        restaurarEscondite()
    end
end

-- Función para escuchar cambios en el atributo "Hiding" del personaje
local function conectarDeteccionPersonaje(character)
    if not character then return end

    if hidingConnection then
        hidingConnection:Disconnect()
        hidingConnection = nil
    end

    hidingConnection = character:GetAttributeChangedSignal("Hiding"):Connect(function()
        actualizarTransparenciaEscondite()
    end)

    actualizarTransparenciaEscondite()
end

    conectarDeteccionPersonaje(LocalPlayer.Character)
end

LocalPlayer.CharacterAdded:Connect(function(newChar)
    conectarDeteccionPersonaje(newChar)
end)

local function registrarHighlight(instancia)
    if instancia:IsA("Highlight") and not highlightsRegistrados[instancia] then
        -- Guardamos la instancia junto a sus colores originales
        highlightsRegistrados[instancia] = {
            FillColor = instancia.FillColor,
            OutlineColor = instancia.OutlineColor
        }

        instancia.Destroying:Connect(function()
            highlightsRegistrados[instancia] = nil
        end)
    end
end

        for _, desc in ipairs(playerGui:GetDescendants()) do
            registrarHighlight(desc)
        end
    end
end

    for _, desc in ipairs(CoreGui:GetDescendants()) do

        if highlight and highlight.Parent then
            highlight.FillColor = coloresOriginales.FillColor
            highlight.OutlineColor = coloresOriginales.OutlineColor

            highlightsRegistrados[highlight] = nil
        end
    end
end
